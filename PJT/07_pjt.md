```python
# weathers/serializers.py

from rest_framework import serializers
from .models import Weather

# Django가 사용하는 데이터 형식 (Python)
# 우리 서비스 = 웹 서비스 -> 다른 언어, 환경에서 요청이 올 수도 있음
#   -> 모든 요청에 부합할 수 있도록 데이터 형식을 정해야 함
#   -> Json 형식으로 돌려주자 : DRF는 Json 형식으로의 변환을 serializers를 통해서 편하게 할 수 있음

# DB에 지정된 필드들만 포장에 관여 : ModelSerializer
#   - 여러 테이블의 데이터를 한 번에 포장 : Nested Serializer
# DB 필드 외의 데이터들도 포장에 관여 : Serializer
class WeatherSerializer(serializers.ModelSerializer):
    class Meta:
        model = Weather
        fields = '__all__'
```

```python
# weathers/views.py

from django.shortcuts import render, redirect
from rest_framework.decorators import api_view
from rest_framework.response import Response
from django.http import JsonResponse
import requests         # pip install requests

from django.conf import settings

from .models import Weather
from .serializers import WeatherSerializer

# 1. OpeWeatherMap API로부터 데이터 다운로드
# 2. 그대로 출력
@api_view(['GET'])
def index(request):
    API_KEY = settings.API_KEY
    CITY_NAME = 'Seoul,KR'
    URL = f'http://api.openweathermap.org/data/2.5/forecast?q={CITY_NAME}&appid={API_KEY}'

    response = requests.get(URL).json()
    return Response(response)

@api_view(['POST'])
def save_data(request):
    # 1. API를 통해 데이터를 가져옴
    API_KEY = settings.API_KEY
    CITY_NAME = 'Seoul,KR'
    URL = f'http://api.openweathermap.org/data/2.5/forecast?q={CITY_NAME}&appid={API_KEY}'
    response = requests.get(URL).json()

    # 2. 원하는 필드만 꺼내서
    for li in response.get('list'):
        dt_txt = li.get('dt_txt')
        temp = li.get('main').get('temp')
        feels_like = li.get('main').get('feels_like')

        # DB에 이미 저장되어 있는지 중복 확인 : 이미 있다면 건너뜀
        if Weather.objects.filter(dt_txt=dt_txt, temp=temp, feels_like=feels_like).exists():
            continue
        
        # 3. DB에 없다면 저장
        #    - 저장하기 위해서 데이터들을 포장해야함 (serializer로 변환)
        #      -> 유효성 검증, 저장 등등 과정을 편하게 다룰 수 있음 
        save_data = {
            'dt_txt': dt_txt,
            'temp': temp,
            'feels_like': feels_like,
        }
        
        serializer = WeatherSerializer(data=save_data)
        if serializer.is_valid(raise_exception=True):
            serializer.save()

    return  JsonResponse({ 'message': '저장 완료' })

@api_view(['GET'])
def list_data(request):
    weathers = Weather.objects.all()
    serializers = WeatherSerializer(weathers, many=True)
    return Response(serializers.data)

@api_view(['GET'])
def hot_weathers(request):
    weathers = Weather.objects.all()
    hot_weathers_list = []
    for weather in weathers:
        # 섭씨 = K - 273.15
        celsius = round(weather.temp - 273.15, 2)
        if celsius > 30:
            hot_weathers_list.append(weather)
    
    serializers = WeatherSerializer(hot_weathers_list, many=True)
    return Response(serializers.data)

@api_view(['GET'])
def cold_weathers(request):
    weathers = Weather.objects.all()
    cold_weathers_list = []
    for weather in weathers:
        # 섭씨 = K - 273.15
        celsius = round(weather.temp - 273.15, 2)
        if celsius < 10:
            cold_weathers_list.append(weather)
    
    serializers = WeatherSerializer(cold_weathers_list, many=True)
    return Response(serializers.data)
```
